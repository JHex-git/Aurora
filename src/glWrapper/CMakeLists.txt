file(GLOB SRCS *.cpp)
set(TARGET glWrapper_lib)
if (MINGW)
    add_library(${TARGET} SHARED ${SRCS})
elseif(MSVC)
    add_library(${TARGET} STATIC ${SRCS})
endif()
target_link_libraries(${TARGET} PRIVATE stb)
target_link_libraries(${TARGET} PUBLIC glad)
if (TARGET shaderc::shaderc_combined)
    target_link_libraries(${TARGET} PRIVATE shaderc::shaderc_combined)
elseif (TARGET shaderc::shaderc)
    target_link_libraries(${TARGET} PRIVATE shaderc::shaderc)
elseif (TARGET shaderc_combined)
    target_link_libraries(${TARGET} PRIVATE shaderc_combined)
elseif (TARGET shaderc)
    target_link_libraries(${TARGET} PRIVATE shaderc)
else()
    message(FATAL_ERROR "shaderc target not found from find_package(shaderc).")
endif()
if (MSVC)
    set_target_properties(${TARGET} PROPERTIES FOLDER "Engine")
endif()
